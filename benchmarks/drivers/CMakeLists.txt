cmake_minimum_required(VERSION 3.22)
project(octree_bench_drivers LANGUAGES CXX CUDA)

# BENCH_VERSION: cstone | v1 | v2 | v3 | check (v3 + cstone leaf-count check)
set(BENCH_VERSION "" CACHE STRING "version to build a driver for")
set(BENCH_SRC "" CACHE PATH "clone of the version")
set(BENCH_CSTONE_SRC "" CACHE PATH "cornerstone clone (check only)")

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 20)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)

if(BENCH_VERSION STREQUAL "cstone" OR BENCH_VERSION STREQUAL "check")
    set(CSTONE_WITH_HIP OFF CACHE BOOL "" FORCE)
    set(CSTONE_WITH_CUDA ON CACHE BOOL "" FORCE)
endif()

if(BENCH_VERSION STREQUAL "cstone")
    add_subdirectory(${BENCH_SRC} cstone)
    add_executable(bench_driver driver_cstone.cu)
    target_link_libraries(bench_driver PRIVATE cstone_gpu)
    target_compile_options(bench_driver PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
elseif(BENCH_VERSION STREQUAL "check")
    add_subdirectory(${BENCH_SRC} v3)
    add_subdirectory(${BENCH_CSTONE_SRC} cstone)
    add_executable(cstone_check cstone_check.cu)
    target_link_libraries(cstone_check PRIVATE adaptive_octree::adaptive_octree cstone_gpu)
    target_compile_options(cstone_check PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--expt-relaxed-constexpr>)
else()
    add_subdirectory(${BENCH_SRC} lib)
    add_executable(bench_driver driver_${BENCH_VERSION}.cu)
    target_link_libraries(bench_driver PRIVATE adaptive_octree::adaptive_octree)
endif()
